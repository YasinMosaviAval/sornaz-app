cmake_minimum_required(VERSION 3.22)
project(sornaz_music_analysis LANGUAGES C CXX)

set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(MA_ENABLE_AUBIO "Enable GPL-licensed Aubio for comparative tests only" OFF)
add_library(music_analysis_dsp SHARED music_analysis_dsp.cpp dsp_engine_open.cpp)
target_include_directories(music_analysis_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MA_ENABLE_AUBIO)
  set(AUBIO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/aubio")
  file(GLOB_RECURSE AUBIO_SOURCES CONFIGURE_DEPENDS "${AUBIO_ROOT}/src/*.c")
  add_library(aubio_vendor STATIC ${AUBIO_SOURCES})
  target_include_directories(aubio_vendor PUBLIC "${AUBIO_ROOT}/src")
  target_compile_definitions(aubio_vendor PRIVATE
    HAVE_STDLIB_H=1 HAVE_STDIO_H=1 HAVE_MATH_H=1 HAVE_STRING_H=1
    HAVE_ERRNO_H=1 HAVE_LIMITS_H=1 HAVE_STDARG_H=1
    HAVE_C99_VARARGS_MACROS=1)
  target_sources(music_analysis_dsp PRIVATE dsp_engine_aubio.cpp)
  target_compile_definitions(music_analysis_dsp PRIVATE MA_ENABLE_AUBIO=1)
  target_link_libraries(music_analysis_dsp PRIVATE aubio_vendor)
endif()
if(NOT MSVC)
  target_link_libraries(music_analysis_dsp PRIVATE m)
  if(MA_ENABLE_AUBIO)
    target_link_libraries(aubio_vendor PUBLIC m)
  endif()
endif()

include(CTest)
if(BUILD_TESTING AND NOT ANDROID AND NOT IOS)
  add_executable(music_analysis_dsp_test tests/music_analysis_dsp_test.cpp)
  target_link_libraries(music_analysis_dsp_test PRIVATE music_analysis_dsp)
  add_test(NAME music_analysis_dsp_test COMMAND music_analysis_dsp_test)
  if(MA_ENABLE_AUBIO)
    add_test(NAME music_analysis_dsp_aubio_test COMMAND music_analysis_dsp_test aubio)
  endif()
  add_executable(music_analysis_c_abi_smoke tests/c_abi_smoke.c)
  target_link_libraries(music_analysis_c_abi_smoke PRIVATE music_analysis_dsp)
  add_test(NAME music_analysis_c_abi_smoke COMMAND music_analysis_c_abi_smoke)
  add_executable(music_analysis_ab_test tests/ab_benchmark.cpp)
  target_link_libraries(music_analysis_ab_test PRIVATE music_analysis_dsp)
  add_test(NAME music_analysis_ab_test COMMAND music_analysis_ab_test)
  add_executable(music_analysis_yin_trace tests/open_dsp_yin_trace.cpp
    music_analysis_dsp.cpp dsp_engine_open.cpp)
  target_include_directories(music_analysis_yin_trace PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(music_analysis_yin_trace PRIVATE MA_TRACE_YIN=1)
  add_executable(music_analysis_octave_synthetic tests/open_dsp_octave_synthetic.cpp)
  target_link_libraries(music_analysis_octave_synthetic PRIVATE music_analysis_dsp)
  add_executable(music_analysis_pitch_candidate_probe
    tests/pitch_candidate_probe.cpp music_analysis_dsp.cpp dsp_engine_open.cpp)
  target_include_directories(music_analysis_pitch_candidate_probe PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(music_analysis_pitch_candidate_probe PRIVATE MA_TRACE_YIN=1)
  add_executable(music_analysis_pitch_baseline_benchmark
    tests/pitch_candidate_baseline_benchmark.cpp)
  target_link_libraries(music_analysis_pitch_baseline_benchmark PRIVATE music_analysis_dsp)
  add_test(NAME music_analysis_octave_synthetic_control
    COMMAND music_analysis_octave_synthetic "${CMAKE_CURRENT_BINARY_DIR}/octave_synthetic_control.csv")
endif()
